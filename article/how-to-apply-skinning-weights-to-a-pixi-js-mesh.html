How to Apply Skinning Weights to a Pixi.js Mesh

Applying skinning weights across multiple bone joints on a Pixi.js Mesh requires extending standard 2D geometry with custom vertex attributes for joint indices and weights, passing bone transformation matrices to a custom shader as uniforms, and calculating the weighted vertex transformations in WebGL. This guide explains how to construct the geometry, write the vertex shader, and update the bone transforms to achieve smooth skeletal deformation.

1. Set Up Custom Mesh Geometry

Pixi.js's standard PIXI.MeshGeometry provides basic attributes like positions and UVs. Linear blend skinning requires two additional per-vertex attributes: the IDs of the bones influencing the vertex and the normalized weights for each bone. Most real-time engines allow up to 4 bone influences per vertex.

import * as PIXI from 'pixi.js';

const geometry = new PIXI.Geometry();

// Standard 2D positions (x, y)
const positions = new Float32Array([
    0, 0,
    100, 0,
    100, 100,
    0, 100
]);

// Indices of the 4 bones influencing each vertex
const jointIndices = new Float32Array([
    0, 1, 0, 0, // Vertex 0 influenced by bone 0 and 1
    0, 1, 0, 0, // Vertex 1 influenced by bone 0 and 1
    1, 2, 0, 0, // Vertex 2 influenced by bone 1 and 2
    1, 2, 0, 0  // Vertex 3 influenced by bone 1 and 2
]);

// Weight/influence of each corresponding bone (must sum to 1.0)
const jointWeights = new Float32Array([
    0.8, 0.2, 0.0, 0.0,
    0.7, 0.3, 0.0, 0.0,
    0.6, 0.4, 0.0, 0.0,
    0.5, 0.5, 0.0, 0.0
]);

const uvs = new Float32Array([0, 0, 1, 0, 1, 1, 0, 1]);
const index = new Uint16Array([0, 1, 2, 0, 2, 3]);

geometry.addAttribute('aVertexPosition', positions, 2)
        .addAttribute('aTextureCoord', uvs, 2)
        .addAttribute('aJointIndices', jointIndices, 4)
        .addAttribute('aJointWeights', jointWeights, 4)
        .addIndex(index);

2. Write the Skinning Vertex Shader

In the vertex shader, read the bone indices and weights, multiply the vertex position by each relevant bone matrix, scale by the weight, and accumulate the result.

// vertex.glsl
precision mediump float;

attribute vec2 aVertexPosition;
attribute vec2 aTextureCoord;
attribute vec4 aJointIndices;
attribute vec4 aJointWeights;

uniform mat3 translationMatrix;
uniform mat3 projectionMatrix;

// Array of 2D affine transform matrices for bones (3x3)
const int MAX_BONES = 32;
uniform mat3 uBoneMatrices[MAX_BONES];

varying vec2 vTextureCoord;

void main(void) {
    vTextureCoord = aTextureCoord;

    vec3 pos = vec3(aVertexPosition, 1.0);
    vec3 skinned = vec3(0.0);

    // Accumulate position based on bone weights
    skinned += (uBoneMatrices[int(aJointIndices.x)] * pos) * aJointWeights.x;
    skinned += (uBoneMatrices[int(aJointIndices.y)] * pos) * aJointWeights.y;
    skinned += (uBoneMatrices[int(aJointIndices.z)] * pos) * aJointWeights.z;
    skinned += (uBoneMatrices[int(aJointIndices.w)] * pos) * aJointWeights.w;

    gl_Position = vec4((projectionMatrix * translationMatrix * vec3(skinned.xy, 1.0)).xy, 0.0, 1.0);
}

The fragment shader remains standard:

// fragment.glsl
precision mediump float;

varying vec2 vTextureCoord;
uniform sampler2D uSampler;

void main(void) {
    gl_FragColor = texture2D(uSampler, vTextureCoord);
}

3. Create the Shader Program and Mesh

Combine the shaders into a PIXI.Shader and pass a flattened array containing the transform matrices for each bone.

const maxBones = 32;
const boneMatrices = new Float32Array(maxBones * 9); // 3x3 matrices

// Initialize identity matrices for each bone
for (let i = 0; i < maxBones; i++) {
    const offset = i * 9;
    boneMatrices[offset] = 1;     // [0,0]
    boneMatrices[offset + 4] = 1; // [1,1]
    boneMatrices[offset + 8] = 1; // [2,2]
}

const shader = PIXI.Shader.from(vertexShaderSource, fragmentShaderSource, {
    uBoneMatrices: boneMatrices,
    uSampler: PIXI.Texture.WHITE
});

const skinnedMesh = new PIXI.Mesh(geometry, shader);
app.stage.addChild(skinnedMesh);

4. Update Bone Matrices at Runtime

Each bone transform must be the combined result of the bone's current world transform multiplied by its inverse bind pose matrix (\(M_{bone} \times M_{bind}^{-1}\)). Update the uniform array on each tick:

function updateBones(bones) {
    for (let i = 0; i < bones.length; i++) {
        const matrix = bones[i].skinMatrix; // Matrix: worldTransform * inverseBindMatrix
        const offset = i * 9;

        // Copy 3x3 matrix values into the uniform array
        boneMatrices[offset + 0] = matrix.a;
        boneMatrices[offset + 1] = matrix.b;
        boneMatrices[offset + 2] = 0;
        boneMatrices[offset + 3] = matrix.c;
        boneMatrices[offset + 4] = matrix.d;
        boneMatrices[offset + 5] = 0;
        boneMatrices[offset + 6] = matrix.tx;
        boneMatrices[offset + 7] = matrix.ty;
        boneMatrices[offset + 8] = 1;
    }
}

app.ticker.add(() => {
    // Update bone hierarchies, then:
    updateBones(skeletonBones);
});
