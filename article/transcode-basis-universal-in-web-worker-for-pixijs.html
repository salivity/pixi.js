Transcode Basis Universal in Web Worker for PixiJS

Offloading Basis Universal texture transcoding to a Web Worker prevents main-thread frame drops and ensures smooth rendering in PixiJS applications. This guide explains how to detect GPU format capabilities, initialize the Basis transcoder inside a dedicated background worker, transcode .basis or .ktx2 files into supported compressed GPU formats, and transfer the resulting buffers directly into PixiJS textures without UI blocking.

1. Detect GPU Capabilities on the Main Thread

The Web Worker cannot directly access the WebGL context, so you must query the supported compressed texture extensions on the main thread and pass these capabilities to the worker.

function getSupportedFormats(renderer) {
  const gl = renderer.gl;
  return {
    astc: !!gl.getExtension('WEBGL_compressed_texture_astc'),
    s3tc: !!gl.getExtension('WEBGL_compressed_texture_s3tc'),
    etc: !!gl.getExtension('WEBGL_compressed_texture_etc'),
    etc1: !!gl.getExtension('WEBGL_compressed_texture_etc1'),
    pvrtc: !!gl.getExtension('WEBGL_compressed_texture_pvrtc')
  };
}

2. Implement the Web Worker

The worker loads the Basis Universal WebAssembly transcoder, parses the incoming .basis binary, selects the optimal target format matching the device's GPU capabilities, and transcodes the image data.

// basis.worker.js
importScripts('basis_transcoder.js');

let transcoderReady = false;

BASIS().then((module) => {
  module.initTranscoder();
  transcoderReady = true;
  postMessage({ type: 'ready' });
});

onmessage = function (e) {
  const { id, buffer, formats } = e.data;

  if (!transcoderReady) {
    postMessage({ id, error: 'Transcoder not initialized' });
    return;
  }

  const { BasisFile, BASIS_FORMAT } = BASIS;
  const basisFile = new BasisFile(new Uint8Array(buffer));

  if (!basisFile.startTranscoding()) {
    basisFile.close();
    basisFile.delete();
    postMessage({ id, error: 'Failed to start transcoding' });
    return;
  }

  // Select target format based on supported hardware capabilities
  let targetFormat = BASIS_FORMAT.cTFRGBA32; // Fallback
  let glInternalFormat = 0x1908; // RGBA

  if (formats.astc) {
    targetFormat = BASIS_FORMAT.cTFASTC_4x4_RGBA;
    glInternalFormat = 0x93B0; // COMPRESSED_RGBA_ASTC_4x4_KHR
  } else if (formats.s3tc) {
    targetFormat = BASIS_FORMAT.cTFBC3_RGBA;
    glInternalFormat = 0x83F0; // COMPRESSED_RGBA_S3TC_DXT5_EXT
  } else if (formats.etc) {
    targetFormat = BASIS_FORMAT.cTFETC2_RGBA;
    glInternalFormat = 0x9278; // COMPRESSED_RGBA8_ETC2_EAC
  }

  const width = basisFile.getImageWidth(0, 0);
  const height = basisFile.getImageHeight(0, 0);
  const dstSize = basisFile.getImageTranscodedSizeInBytes(0, 0, targetFormat);
  const dst = new Uint8Array(dstSize);

  const success = basisFile.transcodeImage(dst, 0, 0, targetFormat, 0, 0);

  basisFile.close();
  basisFile.delete();

  if (!success) {
    postMessage({ id, error: 'Transcode failed' });
    return;
  }

  // Transfer the underlying ArrayBuffer back with zero copy
  postMessage(
    {
      id,
      width,
      height,
      glInternalFormat,
      buffer: dst.buffer
    },
    [dst.buffer]
  );
};

3. Coordinate Work and Ingest into PixiJS

Send raw file buffers to the worker using Transferable Objects. Once returned, map the transcoded binary data to PixiJS via CompressedSource (PixiJS v8) or CompressedTextureResource (PixiJS v7).

PixiJS v8 Implementation

import { Texture, CompressedSource } from 'pixi.js';

class BasisLoader {
  constructor(renderer, workerUrl) {
    this.worker = new Worker(workerUrl);
    this.formats = getSupportedFormats(renderer);
    this.callbacks = new Map();
    this.reqId = 0;

    this.worker.onmessage = (e) => {
      const { id, width, height, glInternalFormat, buffer, error } = e.data;
      if (this.callbacks.has(id)) {
        const { resolve, reject } = this.callbacks.get(id);
        this.callbacks.delete(id);

        if (error) {
          reject(new Error(error));
        } else {
          resolve({ width, height, glInternalFormat, buffer });
        }
      }
    };
  }

  async load(url) {
    const response = await fetch(url);
    const arrayBuffer = await response.arrayBuffer();

    return new Promise((resolve, reject) => {
      const id = ++this.reqId;
      this.callbacks.set(id, {
        resolve: (transcoded) => {
          const source = new CompressedSource({
            resource: new Uint8Array(transcoded.buffer),
            width: transcoded.width,
            height: transcoded.height,
            format: transcoded.glInternalFormat
          });

          const texture = new Texture({ source });
          resolve(texture);
        },
        reject
      });

      // Transfer ownership of arrayBuffer to worker
      this.worker.postMessage(
        { id, buffer: arrayBuffer, formats: this.formats },
        [arrayBuffer]
      );
    });
  }
}

PixiJS v7 Implementation

In PixiJS v7, wrap the returned buffer using CompressedTextureResource:

import * as PIXI from 'pixi.js';

function createTextureV7(transcodedData) {
  const resource = new PIXI.CompressedTextureResource(
    new Uint8Array(transcodedData.buffer),
    {
      format: transcodedData.glInternalFormat,
      width: transcodedData.width,
      height: transcodedData.height,
      levels: 1
    }
  );

  const baseTexture = new PIXI.BaseTexture(resource);
  return new PIXI.Texture(baseTexture);
}

Summary of Data Flow

  1. Main Thread: Checks WebGL format support and fetches the compressed asset.
  2. Worker Thread: Receives the raw .basis bytes, executes the WASM transcoder, and writes the native GPU texture format.
  3. Zero-Copy Transfer: The transcoded ArrayBuffer is passed back via postMessage transfer list.
  4. Main Thread GPU Upload: PixiJS wraps the buffer into a compressed texture resource and binds it directly to the GPU without parsing overhead.
