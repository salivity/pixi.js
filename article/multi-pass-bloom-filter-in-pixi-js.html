Multi-Pass Bloom Filter in Pixi.js

This article explains how to build a high-performance multi-pass bloom filter in Pixi.js utilizing progressive downsampling and upsampling chains. By extracting high-luminance areas, downsampling them through successive mip levels with a box filter, and progressively upsampling with additive blending, this approach achieves smooth, wide-radius cinematic glows while minimizing texture bandwidth and GPU overhead.

Overview of the Pipeline

A multi-pass bloom pipeline avoids the performance bottlenecks of large-kernel Gaussian blurs. Instead, it computes bloom in four stages:

  1. Luminance Threshold (Prefilter): Isolates the bright regions of the scene.
  2. Downsample Chain: Successively halves the resolution across multiple render textures (typically 4 to 6 iterations), using an anti-flicker 13-tap filter.
  3. Upsample Chain: Progressively increases resolution while additively combining each mip level with a tent filter.
  4. Composite: Blends the final bloom texture back over the original scene texture.

Shader Definitions

The implementation requires four small GLSL fragment shaders. The vertex shader remains the default Pixi.js pass-through.

const defaultVertex = `
    attribute vec2 aVertexPosition;
    attribute vec2 aTextureCoord;
    uniform mat3 projectionMatrix;
    varying vec2 vTextureCoord;

    void main(void) {
        gl_Position = vec4((projectionMatrix * vec3(aVertexPosition, 1.0)).xy, 0.0, 1.0);
        vTextureCoord = aTextureCoord;
    }
`;

// 1. Extract bright spots
const thresholdFragment = `
    varying vec2 vTextureCoord;
    uniform sampler2D uSampler;
    uniform float uThreshold;
    uniform float uSoftKnee;

    void main(void) {
        vec4 color = texture2D(uSampler, vTextureCoord);
        float brightness = max(color.r, max(color.g, color.b));
        
        // Soft knee thresholding
        float knee = uThreshold * uSoftKnee;
        float soft = brightness - uThreshold + knee;
        soft = clamp(soft, 0.0, 2.0 * knee);
        soft = (soft * soft) / (4.0 * knee + 0.00001);
        
        float contribution = max(soft, brightness - uThreshold);
        contribution /= max(brightness, 0.00001);
        
        gl_FragColor = vec4(color.rgb * contribution, color.a);
    }
`;

// 2. 13-tap downsampling filter (prevents fireflies/flickering)
const downsampleFragment = `
    varying vec2 vTextureCoord;
    uniform sampler2D uSampler;
    uniform vec2 uTexelSize;

    void main(void) {
        vec2 uv = vTextureCoord;
        vec2 d = uTexelSize;

        // Center box
        vec4 a = texture2D(uSampler, uv);
        // Outer samples
        vec4 b = texture2D(uSampler, uv + vec2(-2.0, -2.0) * d);
        vec4 c = texture2D(uSampler, uv + vec2( 0.0, -2.0) * d);
        vec4 dSample = texture2D(uSampler, uv + vec2( 2.0, -2.0) * d);
        vec4 e = texture2D(uSampler, uv + vec2(-2.0,  0.0) * d);
        vec4 f = texture2D(uSampler, uv + vec2( 2.0,  0.0) * d);
        vec4 g = texture2D(uSampler, uv + vec2(-2.0,  2.0) * d);
        vec4 h = texture2D(uSampler, uv + vec2( 0.0,  2.0) * d);
        vec4 i = texture2D(uSampler, uv + vec2( 2.0,  2.0) * d);
        // Inner samples
        vec4 j = texture2D(uSampler, uv + vec2(-1.0, -1.0) * d);
        vec4 k = texture2D(uSampler, uv + vec2( 1.0, -1.0) * d);
        vec4 l = texture2D(uSampler, uv + vec2(-1.0,  1.0) * d);
        vec4 m = texture2D(uSampler, uv + vec2( 1.0,  1.0) * d);

        vec4 result = e * 0.125 + f * 0.125 + c * 0.125 + h * 0.125;
        result += (j + k + l + m) * 0.5;
        result += (b + dSample + g + i) * 0.0625;

        gl_FragColor = result * 0.5;
    }
`;

// 3. 9-tap tent upsampling filter
const upsampleFragment = `
    varying vec2 vTextureCoord;
    uniform sampler2D uSampler;
    uniform vec2 uTexelSize;
    uniform float uRadius;

    void main(void) {
        vec2 d = uTexelSize * uRadius;

        vec4 result = texture2D(uSampler, vTextureCoord) * 4.0;
        result += texture2D(uSampler, vTextureCoord + vec2(-d.x, -d.y));
        result += texture2D(uSampler, vTextureCoord + vec2( 0.0, -d.y)) * 2.0;
        result += texture2D(uSampler, vTextureCoord + vec2( d.x, -d.y));
        result += texture2D(uSampler, vTextureCoord + vec2(-d.x,  0.0)) * 2.0;
        result += texture2D(uSampler, vTextureCoord + vec2( d.x,  0.0)) * 2.0;
        result += texture2D(uSampler, vTextureCoord + vec2(-d.x,  d.y));
        result += texture2D(uSampler, vTextureCoord + vec2( 0.0,  d.y)) * 2.0;
        result += texture2D(uSampler, vTextureCoord + vec2( d.x,  d.y));

        gl_FragColor = result * (1.0 / 16.0);
    }
`;

// 4. Final Composite
const compositeFragment = `
    varying vec2 vTextureCoord;
    uniform sampler2D uSampler;       // Original scene
    uniform sampler2D uBloomTexture;  // Bloom result
    uniform float uBloomIntensity;

    void main(void) {
        vec4 base = texture2D(uSampler, vTextureCoord);
        vec4 bloom = texture2D(uBloomTexture, vTextureCoord);
        gl_FragColor = base + (bloom * uBloomIntensity);
    }
`;

Implementing the MultiPassBloomFilter

The custom filter class inherits from PIXI.Filter and manages dynamic render targets across its processing passes.

class MultiPassBloomFilter extends PIXI.Filter {
    constructor(options = {}) {
        super(defaultVertex, compositeFragment);

        this.passes = options.passes || 5;
        this.threshold = options.threshold ?? 0.8;
        this.softKnee = options.softKnee ?? 0.5;
        this.bloomRadius = options.bloomRadius ?? 1.0;
        this.intensity = options.intensity ?? 1.2;

        this.uniforms.uBloomIntensity = this.intensity;

        // Sub-filters
        this.thresholdFilter = new PIXI.Filter(defaultVertex, thresholdFragment, {
            uThreshold: this.threshold,
            uSoftKnee: this.softKnee,
        });

        this.downsampleFilter = new PIXI.Filter(defaultVertex, downsampleFragment, {
            uTexelSize: new Float32Array([0, 0]),
        });

        this.upsampleFilter = new PIXI.Filter(defaultVertex, upsampleFragment, {
            uTexelSize: new Float32Array([0, 0]),
            uRadius: this.bloomRadius,
        });
        this.upsampleFilter.blendMode = PIXI.BLEND_MODES.ADD;

        // Texture chains
        this.downsampleTargets = [];
        this.upsampleTargets = [];
    }

    initTargets(width, height) {
        this.destroyTargets();

        let w = Math.floor(width / 2);
        let h = Math.floor(height / 2);

        for (let i = 0; i < this.passes; i++) {
            w = Math.max(1, w);
            h = Math.max(1, h);

            this.downsampleTargets.push(
                PIXI.RenderTexture.create({ width: w, height: h, scaleMode: PIXI.SCALE_MODES.LINEAR })
            );
            this.upsampleTargets.push(
                PIXI.RenderTexture.create({ width: w, height: h, scaleMode: PIXI.SCALE_MODES.LINEAR })
            );

            w = Math.floor(w / 2);
            h = Math.floor(h / 2);
        }
    }

    destroyTargets() {
        this.downsampleTargets.forEach(t => t.destroy(true));
        this.upsampleTargets.forEach(t => t.destroy(true));
        this.downsampleTargets = [];
        this.upsampleTargets = [];
    }

    apply(filterManager, input, output, clearMode) {
        const width = input.filterFrame.width;
        const height = input.filterFrame.height;

        if (!this.downsampleTargets.length || 
            this.downsampleTargets[0].width !== Math.floor(width / 2) || 
            this.downsampleTargets[0].height !== Math.floor(height / 2)) {
            this.initTargets(width, height);
        }

        const renderer = filterManager.renderer;

        // Step 1: Extract bright areas into downsampleTargets[0]
        this.thresholdFilter.uniforms.uThreshold = this.threshold;
        this.thresholdFilter.uniforms.uSoftKnee = this.softKnee;
        renderer.render(input, {
            renderTexture: this.downsampleTargets[0],
            clear: true,
            filter: this.thresholdFilter,
        });

        // Step 2: Downsample chain
        let currentSource = this.downsampleTargets[0];
        for (let i = 1; i < this.passes; i++) {
            const target = this.downsampleTargets[i];
            this.downsampleFilter.uniforms.uTexelSize[0] = 1.0 / currentSource.width;
            this.downsampleFilter.uniforms.uTexelSize[1] = 1.0 / currentSource.height;

            filterManager.applyFilter(this.downsampleFilter, currentSource, target, PIXI.CLEAR_MODES.CLEAR);
            currentSource = target;
        }

        // Step 3: Upsample chain with additive blending
        let lastUpsample = this.downsampleTargets[this.passes - 1];
        this.upsampleFilter.uniforms.uRadius = this.bloomRadius;

        for (let i = this.passes - 2; i >= 0; i--) {
            const target = this.upsampleTargets[i];
            const downsampleCounterpart = this.downsampleTargets[i];

            // Render current level to upsample target
            this.upsampleFilter.uniforms.uTexelSize[0] = 1.0 / lastUpsample.width;
            this.upsampleFilter.uniforms.uTexelSize[1] = 1.0 / lastUpsample.height;
            
            // First copy the downsample pass into upsample target
            renderer.render(downsampleCounterpart, {
                renderTexture: target,
                clear: true,
            });

            // Add the upsampled lower mip over it
            filterManager.applyFilter(this.upsampleFilter, lastUpsample, target, PIXI.CLEAR_MODES.BLEND);
            lastUpsample = target;
        }

        // Step 4: Final composite pass over the original output
        this.uniforms.uBloomTexture = lastUpsample;
        this.uniforms.uBloomIntensity = this.intensity;
        filterManager.applyFilter(this, input, output, clearMode);
    }
}

Using the Filter

Add the custom bloom filter to any Pixi.js container or display object:

const app = new PIXI.Application({
    width: 1280,
    height: 720,
    backgroundColor: 0x050505,
});
document.body.appendChild(app.view);

const bloomFilter = new MultiPassBloomFilter({
    passes: 5,
    threshold: 0.7,
    softKnee: 0.4,
    bloomRadius: 1.2,
    intensity: 1.5,
});

const sceneContainer = new PIXI.Container();
sceneContainer.filters = [bloomFilter];
app.stage.addChild(sceneContainer);

// Add bright elements to demonstrate bloom
const glowCircle = new PIXI.Graphics();
glowCircle.beginFill(0xffffff);
glowCircle.drawCircle(300, 360, 40);
glowCircle.endFill();

const neonSquare = new PIXI.Graphics();
neonSquare.beginFill(0x00ffcc);
neonSquare.drawRect(500, 300, 120, 120);
neonSquare.endFill();

sceneContainer.addChild(glowCircle, neonSquare);

Performance Considerations

  • Pass Count: 4 to 6 passes typically provide a wide diffusion. Adding more passes beyond 6 delivers diminishing visual returns while increasing state changes and render target swaps.
  • Texture Reallocation: Render targets are lazily allocated inside apply() and only regenerated when the viewport resolution changes, keeping runtime frame allocations at zero.
  • Texture Clean-up: Always call destroyTargets() when removing or switching away from the filter to prevent WebGL frame buffer leaks.
